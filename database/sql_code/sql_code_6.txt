CREATE POLICY "Anyone can view follows"
ON public.follows
FOR SELECT
TO anon, authenticated
USING (true);

CREATE POLICY "Users can follow others"
ON public.follows
FOR INSERT
TO authenticated
WITH CHECK (auth.uid() = follower_id);

CREATE POLICY "Users can unfollow"
ON public.follows
FOR DELETE
TO authenticated
USING (auth.uid() = follower_id);
