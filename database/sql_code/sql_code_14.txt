-- Bảng sự kiện nghe nhạc đã được bộ đệm đánh giá.
-- Mỗi history_id chỉ có một kết quả đánh giá hiện hành.
CREATE TABLE public.ranking_events (
    history_id BIGINT PRIMARY KEY
        REFERENCES public.listening_history(history_id) ON DELETE CASCADE,
    user_id UUID NOT NULL
        REFERENCES public.profiles(user_id) ON DELETE CASCADE,
    song_id BIGINT NOT NULL
        REFERENCES public.songs(song_id) ON DELETE CASCADE,
    listened_seconds INTEGER NOT NULL DEFAULT 0
        CHECK (listened_seconds >= 0),
    completed BOOLEAN NOT NULL DEFAULT FALSE,
    status TEXT NOT NULL
        CHECK (status IN ('accepted', 'suspicious', 'rejected')),
    reason TEXT,
    evaluated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX ranking_events_status_song_idx
    ON public.ranking_events(status, song_id);

CREATE INDEX ranking_events_user_song_time_idx
    ON public.ranking_events(user_id, song_id, evaluated_at DESC);

ALTER TABLE public.ranking_events ENABLE ROW LEVEL SECURITY;

-- Backend dùng SUPABASE_SECRET_KEY cần đọc dữ liệu nguồn để đánh giá lượt nghe.
GRANT SELECT ON TABLE public.listening_history TO service_role;
GRANT SELECT ON TABLE public.songs TO service_role;

-- Chỉ backend dùng SUPABASE_SECRET_KEY được phép đọc/ghi sự kiện đã đánh giá.
REVOKE ALL ON TABLE public.ranking_events FROM PUBLIC, anon, authenticated;
GRANT ALL ON TABLE public.ranking_events TO service_role;

-- Bảng tổng hợp lượt nghe hợp lệ; chỉ công khai số liệu xếp hạng.
CREATE TABLE public.song_rankings (
    song_id BIGINT PRIMARY KEY
        REFERENCES public.songs(song_id) ON DELETE CASCADE,
    play_count BIGINT NOT NULL DEFAULT 0
        CHECK (play_count >= 0),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.song_rankings ENABLE ROW LEVEL SECURITY;

GRANT SELECT ON TABLE public.song_rankings TO anon, authenticated;
GRANT ALL ON TABLE public.song_rankings TO service_role;

CREATE POLICY "Public can view song rankings"
    ON public.song_rankings
    FOR SELECT
    TO anon, authenticated
    USING (TRUE);

-- Khởi tạo một dòng xếp hạng cho mỗi bài hát hiện có.
-- Chỉ lượt nghe đạt tiêu chí mới được tính vào play_count.
WITH recent_plays AS (
    SELECT
        history_id,
        COUNT(*) OVER (
            PARTITION BY user_id, song_id
            ORDER BY played_at
            RANGE BETWEEN INTERVAL '60 seconds' PRECEDING AND CURRENT ROW
        ) AS plays_in_60_seconds
    FROM public.listening_history
),
classified_history AS (
    SELECT
        history.history_id,
        history.user_id,
        history.song_id,
        COALESCE(history.listened_seconds, 0) AS listened_seconds,
        COALESCE(history.completed, FALSE) AS completed,
        CASE
            WHEN song.duration_seconds IS NOT NULL
                 AND COALESCE(history.listened_seconds, 0) > song.duration_seconds + 5
                THEN 'rejected'
            WHEN recent.plays_in_60_seconds > 5
                THEN 'suspicious'
            WHEN COALESCE(history.completed, FALSE)
                 OR COALESCE(history.listened_seconds, 0) >=
                    CASE
                        WHEN song.duration_seconds IS NULL THEN 30
                        ELSE LEAST(30, CEIL(song.duration_seconds * 0.8)::INTEGER)
                    END
                THEN 'accepted'
            ELSE 'suspicious'
        END AS status
    FROM public.listening_history AS history
    JOIN public.songs AS song
        ON song.song_id = history.song_id
    JOIN recent_plays AS recent
        ON recent.history_id = history.history_id
)
INSERT INTO public.ranking_events (
    history_id,
    user_id,
    song_id,
    listened_seconds,
    completed,
    status,
    reason
)
SELECT
    history_id,
    user_id,
    song_id,
    listened_seconds,
    completed,
    status,
    CASE
        WHEN status = 'accepted' THEN NULL
        WHEN status = 'rejected' THEN 'listened_time_exceeds_song_duration'
        ELSE 'not_enough_listening_time_or_suspicious_frequency'
    END
FROM classified_history
ON CONFLICT (history_id) DO NOTHING;

INSERT INTO public.song_rankings (song_id, play_count)
SELECT
    song.song_id,
    COUNT(event.history_id) FILTER (WHERE event.status = 'accepted')::BIGINT
FROM public.songs AS song
LEFT JOIN public.ranking_events AS event
    ON event.song_id = song.song_id
GROUP BY song.song_id
ON CONFLICT (song_id) DO UPDATE
SET
    play_count = EXCLUDED.play_count,
    updated_at = NOW();
